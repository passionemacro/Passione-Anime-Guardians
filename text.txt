local Http = game:GetService("HttpService")

local function Encode(data)
	return Http:JSONEncode(data)
end

local function Decode(data)
	return Http:JSONDecode(data)
end

	makefolder("Passione")
end

    Accent = Color3.fromHex("#18181b"),

        ["0"] = { Color = Color3.fromHex("#ba3fba"), Transparency = 0 },            
        ["100"]   = { Color = Color3.fromHex("#000088"), Transparency = 0 },        
    }, {                                                                            
        Rotation = 45,                                                               
    }),
    Outline = Color3.fromHex("#FFFFFF"),
    Text = Color3.fromHex("#FFFFFF"),

local Window = WindUI:CreateWindow({
    Title = "Anime Guardians",
    Icon = "door-open", -- lucide icon
    Author = "Passione",
    Folder = "Passione",
    
    -- ↓ This all is Optional. You can remove it.
    Size = UDim2.fromOffset(700, 460),
    MinSize = Vector2.new(560, 350),
    MaxSize = Vector2.new(850, 560),

local GameTab = Window:Tab({
    Title = "Game",
    Icon = "gamepad-2", -- optional
    Locked = false,
})
local MacroTab = Window:Tab({
    Title = "Macro",
    Icon = "play", -- optional
    Locked = false,
})
local StepTracker = MacroTab:Paragraph({
    Title = "Steps",
    Desc = [[
Unit:
Action:
Cost:
Waiting For: ]],
    Image = "",
    ImageSize = 30,
    Thumbnail = "",
    ThumbnailSize = 80,
    Locked = false,
})
local MacroDropdown = MacroTab:Dropdown({
    Title = "Advanced Dropdown",
    Desc = "Dropdown Description",
    Values = {},
    Value = nil,
    Callback = function(option) 
        print("Category selected: " .. option.Title .. " with icon " .. option.Icon) 
    end
})
local MacroNameInput = MacroTab:Input({
    Title = "Macro Name",
    Desc = "Name of the macro file.",
    Value = nil,
    InputIcon = "folder-pen",
    Type = "Input", -- or "Textarea"
    Placeholder = "Enter Macro Name",
    Callback = function(input)
		if input == nil or input == "" then 
			return 
		end
		local dropdowninput = {
		Title = input,
		Icon = " ",
		}
		table.insert(MacroDropdown.Values, dropdowninput)
		MacroDropdown:Refresh(MacroDropdown.Values)
		WindUI:Notify({
    		Title = "Created File",
    		Content = "Created file: " .. input,
    		Duration = 3, -- 3 seconds
    		Icon = "check",
		})
		writefile("Passione/" .. input .. ".json", " ")
   end
})
local DeleteMacroButton = MacroTab:Button({
    Title = "Delete selected macro",
    Color = Color3.fromHex("ffffff"), -- paint the button
    Justify = "Center", -- align items in the center (Center or Between or Left or Right)
    IconAlign = "Left", -- Left or Right of the text
    Icon = "", -- removing icon
    Callback = function()
        WindUI:Popup({
    Title = "Delete Macro",
    Icon = "info",
    Content = "Are you sure you want to delete the macro file?",
    Buttons = {
        {
            Title = "Cancel",
            Callback = function() end,
            Variant = "Tertiary",
        },
        {
            Title = "Continue",
            Icon = "arrow-right",
            Callback = function() 
					local selected = MacroDropdown.Value
					if not selected then return end
					
					for i,v in ipairs(MacroDropdown.Values) do
						if v.Title == selected.Title then
							table.remove(MacroDropdown.Values, i)
							delfile("Passione/" .. selected.Title .. ".json")
							break
						end
					end
					MacroDropdown:Refresh(MacroDropdown.Values)
				end,
            Variant = "Primary",
        }
    }
})
    end
})
